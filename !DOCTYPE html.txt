        /**
         * ESTADO DE LA APLICACIÓN Y GESTIÓN DE LOCALSTORAGE
         */
        const STORAGE_KEY = 'clinidle_interview_v2';
        const MAX_SESSION_MINUTES = 45;
        const MINUTES_PER_ACTION = 5;
        const MAX_DIAGNOSIS_ATTEMPTS = 3;

        let appState = {
            currentCaseId: 1,
            stats: {
                played: 0,
                won: 0,
                currentStreak: 0,
                maxStreak: 0
            },
            // Mapa por ID de caso:
            // { askedQuestionIds: [], failedGuesses: [], timeLeft: 45, status: 'playing'|'won'|'lost' }
            casesProgress: {}
        };

        let selectedDisorder = null;
        let activeTabArea = 'anamnesis';
        let questionFilterText = '';

        function loadState() {
            try {
                const saved = localStorage.getItem(STORAGE_KEY);
                if (saved) {
                    const parsed = JSON.parse(saved);
                    appState = {
                        ...appState,
                        ...parsed,
                        stats: { ...appState.stats, ...(parsed.stats || {}) },
                        casesProgress: parsed.casesProgress || {}
                    };
                }
            } catch (e) {
                console.warn("No se pudo leer localStorage:", e);
            }

            CLINICAL_CASES.forEach(c => {
                if (!appState.casesProgress[c.id]) {
                    appState.casesProgress[c.id] = {
                        askedQuestionIds: [],
                        failedGuesses: [],
                        timeLeft: MAX_SESSION_MINUTES,
                        status: 'playing'
                    };
                }
            });
        }

        function saveState() {
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
            } catch (e) {
                console.warn("No se pudo guardar en localStorage:", e);
            }
        }

        /**
         * UTILIDADES UI
         */
        let toastTimeout = null;
        function showToast(message, type = 'success') {
            const container = document.getElementById('toast-container');
            const content = document.getElementById('toast-content');
            const msgEl = document.getElementById('toast-message');

            msgEl.textContent = message;
            if (type === 'error') {
                content.className = "bg-rose-950 border border-rose-700 text-rose-100 px-4 py-3 rounded-xl shadow-2xl flex items-center gap-2.5 text-sm font-medium";
            } else if (type === 'key') {
                content.className = "bg-amber-950 border border-amber-500/60 text-amber-100 px-4 py-3 rounded-xl shadow-2xl flex items-center gap-2.5 text-sm font-medium";
            } else {
                content.className = "bg-slate-800 border border-emerald-500/40 text-slate-100 px-4 py-3 rounded-xl shadow-2xl flex items-center gap-2.5 text-sm font-medium";
            }

            container.classList.remove('opacity-0', 'translate-y-4');
            container.classList.add('opacity-100', 'translate-y-0');

            clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                container.classList.remove('opacity-100', 'translate-y-0');
                container.classList.add('opacity-0', 'translate-y-4');
            }, 3200);
        }

        function normalizeText(str) {
            return str.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toLowerCase();
        }

        function getCurrentCase() {
            return CLINICAL_CASES.find(c => c.id === appState.currentCaseId) || CLINICAL_CASES[0];
        }

        function getCurrentProgress() {
            return appState.casesProgress[appState.currentCaseId];
        }

        function getDisorderById(id) {
            return DSM_DISORDERS.find(d => d.id === id);
        }

        /**
         * RENDERIZADO DE COMPONENTES DE LA INTERFAZ
         */
        function renderCaseSelector() {
            const selector = document.getElementById('case-selector');
            selector.innerHTML = '';

            CLINICAL_CASES.forEach(c => {
                const prog = appState.casesProgress[c.id];
                let statusIcon = '';
                if (prog.status === 'won') statusIcon = '✅ ';
                else if (prog.status === 'lost') statusIcon = '❌ ';

                const opt = document.createElement('option');
                opt.value = c.id;
                opt.textContent = `${statusIcon}${c.label}`;
                if (c.id === appState.currentCaseId) opt.selected = true;
                selector.appendChild(opt);
            });
        }

        function renderHeaderAndTriage() {
            const currentCase = getCurrentCase();
            const progress = getCurrentProgress();

            document.getElementById('streak-counter').textContent = appState.stats.currentStreak;
            document.getElementById('case-badge').textContent = `#0${currentCase.id}`;
            document.getElementById('case-title').textContent = `Expediente Clínico #0${currentCase.id}`;

            const dailyTag = document.getElementById('daily-tag');
            if (currentCase.isDaily) {
                dailyTag.textContent = "Caso del Día";
                dailyTag.className = "text-[11px] font-semibold px-2 py-0.5 rounded-full bg-emerald-500/15 text-emerald-400 border border-emerald-500/30";
            } else {
                dailyTag.textContent = "Archivo Clínico";
                dailyTag.className = "text-[11px] font-semibold px-2 py-0.5 rounded-full bg-slate-800 text-slate-300 border border-slate-700";
            }

            // Tiempo restante de consulta
            const timeEl = document.getElementById('session-time-text');
            const clockIcon = document.getElementById('clock-icon');
            timeEl.textContent = `${progress.timeLeft} min`;
            if (progress.timeLeft <= 10) {
                timeEl.className = "text-sm font-mono font-bold text-rose-400 mt-0.5 block";
                clockIcon.className = "w-4 h-4 text-rose-400";
            } else if (progress.timeLeft <= 20) {
                timeEl.className = "text-sm font-mono font-bold text-amber-400 mt-0.5 block";
                clockIcon.className = "w-4 h-4 text-amber-400";
            } else {
                timeEl.className = "text-sm font-mono font-bold text-emerald-400 mt-0.5 block";
                clockIcon.className = "w-4 h-4 text-emerald-400";
            }

            // 3 Puntos de Hipótesis Diagnóstica
            const dotsContainer = document.getElementById('attempts-dots');
            dotsContainer.innerHTML = '';
            const targetDisorder = getDisorderById(currentCase.targetId);

            for (let i = 0; i < MAX_DIAGNOSIS_ATTEMPTS; i++) {
                const dot = document.createElement('div');
                const failedId = progress.failedGuesses[i];
                if (failedId) {
                    const guessed = getDisorderById(failedId);
                    if (guessed && guessed.category === targetDisorder.category) {
                        dot.className = "w-3.5 h-3.5 rounded-sm bg-amber-400";
                        dot.title = "Misma categoría diagnóstica";
                    } else {
                        dot.className = "w-3.5 h-3.5 rounded-sm bg-rose-500";
                        dot.title = "Categoría distinta";
                    }
                } else if (progress.status === 'won' && i === progress.failedGuesses.length) {
                    dot.className = "w-3.5 h-3.5 rounded-sm bg-emerald-500 shadow-sm shadow-emerald-500/50";

        function calculateSessionMetrics() {
            const currentCase = getCurrentCase();
            const progress = getCurrentProgress();

            const timeUsed = MAX_SESSION_MINUTES - progress.timeLeft;
            const totalKey = currentCase.questions.filter(q => q.isKey).length;
            const foundKey = currentCase.questions.filter(
                q => q.isKey && progress.askedQuestionIds.includes(q.id)
            ).length;

            // Cálculo de eficiencia clínica (0 - 100%)
            let efficiency = 0;
            if (progress.status === 'won') {
                const clueRatio = totalKey > 0 ? (foundKey / totalKey) : 1;
                const timeBonus = Math.max(0, (progress.timeLeft / MAX_SESSION_MINUTES) * 35);
                const attemptPenalty = progress.failedGuesses.length * 12;
                efficiency = Math.min(100, Math.max(40, Math.round((clueRatio * 65) + timeBonus - attemptPenalty)));
            } else {
                efficiency = Math.round((foundKey / Math.max(1, totalKey)) * 35);
            }

            let rank = "En Formación";
            if (progress.status === 'won') {
                if (efficiency >= 85) rank = "🏅 Clínico Experto";
                else if (efficiency >= 65) rank = "🩺 Residente Destacado";
                else rank = "📋 Diagnóstico Certero";
            } else {
                rank = "📚 Caso de Estudio";
            }

            return { timeUsed, totalKey, foundKey, efficiency, rank };
        }

        function generateShareText() {
            const currentCase = getCurrentCase();
            const progress = getCurrentProgress();
            const targetDisorder = getDisorderById(currentCase.targetId);
            const metrics = calculateSessionMetrics();

            const boxes = progress.failedGuesses.map(id => {
                const d = getDisorderById(id);
                return d && d.category === targetDisorder.category ? '🟨' : '⬜';
            });

            if (progress.status === 'won') {
                boxes.push('🟩');
            }
            while (boxes.length < MAX_DIAGNOSIS_ATTEMPTS) {
                boxes.push('▫️');
            }

            return `🏥 CliniDle #0${currentCase.id} | ⏱️ ${metrics.timeUsed}m | 🔍 ${metrics.foundKey}/${metrics.totalKey} | ${boxes.join('')} | 🔥 ${appState.stats.currentStreak}`;
        }

        function askClinicalQuestion(questionId) {
            const currentCase = getCurrentCase();
            const progress = getCurrentProgress();
            if (progress.status !== 'playing') return;
            if (progress.askedQuestionIds.includes(questionId)) return;

            if (progress.timeLeft < MINUTES_PER_ACTION) {
                showToast("No queda tiempo de consulta. Emite tu diagnóstico definitivo.", "error");
                return;
            }

            const questionObj = currentCase.questions.find(q => q.id === questionId);
            if (!questionObj) return;

            progress.askedQuestionIds.push(questionId);
            progress.timeLeft = Math.max(0, progress.timeLeft - MINUTES_PER_ACTION);

            if (questionObj.isKey) {
                showToast(`¡${questionObj.tagType} añadido a tu bitácora!`, "key");
            } else if (progress.timeLeft === 0) {
                showToast("Tiempo de consulta agotado (45 min). Formula tu diagnóstico.", "error");
            }

            saveState();
            renderAll();
        }

        function handleGuessSubmission() {
            if (!selectedDisorder) return;
            const currentCase = getCurrentCase();
            const progress = getCurrentProgress();
            if (progress.status !== 'playing') return;

            const guessedId = selectedDisorder.id;

            // Limpiar input
            searchInput.value = '';
            selectedDisorder = null;
            submitBtn.disabled = true;
            clearBtn.classList.add('hidden');

            if (guessedId === currentCase.targetId) {
                progress.status = 'won';
                appState.stats.played += 1;
                appState.stats.won += 1;
                appState.stats.currentStreak += 1;
                if (appState.stats.currentStreak > appState.stats.maxStreak) {
                    appState.stats.maxStreak = appState.stats.currentStreak;
                }
                showToast("¡Diagnóstico clínico acertado!");
            } else {
                progress.failedGuesses.push(guessedId);
                progress.timeLeft = Math.max(0, progress.timeLeft - MINUTES_PER_ACTION);

                if (progress.failedGuesses.length >= MAX_DIAGNOSIS_ATTEMPTS) {
                    progress.status = 'lost';
                    appState.stats.played += 1;
                    appState.stats.currentStreak = 0;
                    showToast("Se agotaron los 3 intentos de diagnóstico.", "error");
                } else {
                    const targetDisorder = getDisorderById(currentCase.targetId);
                    const guessedDisorder = getDisorderById(guessedId);
                    if (guessedDisorder.category === targetDisorder.category) {
                        showToast("¡Misma categoría diagnóstica! Revisa los criterios diferenciales (-5 min).", "key");
                    } else {
                        showToast("Diagnóstico incorrecto (-5 min de consulta).", "error");
                    }
                }
            }

            saveState();
            renderAll();
            if (progress.status !== 'playing') {
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }
        }

        function handleConcludeCase() {
            const progress = getCurrentProgress();
            if (progress.status !== 'playing') return;

            progress.status = 'lost';
            appState.stats.played += 1;
            appState.stats.currentStreak = 0;
            saveState();
            renderAll();
            showToast("Consulta cerrada. Revisa la supervisión clínica.", "error");
            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

        const modalHelp = document.getElementById('modal-help');
        const modalStats = document.getElementById('modal-stats');

        document.getElementById('btn-help').addEventListener('click', () => {
            modalHelp.classList.remove('hidden');
        });

        document.getElementById('btn-stats').addEventListener('click', () => {
            document.getElementById('stat-played').textContent = appState.stats.played;
            const winRate = appState.stats.played > 0
                ? Math.round((appState.stats.won / appState.stats.played) * 100)
                : 0;
            document.getElementById('stat-winrate').textContent = `${winRate}%`;
            document.getElementById('stat-streak').textContent = appState.stats.currentStreak;
            document.getElementById('stat-maxstreak').textContent = appState.stats.maxStreak;
            modalStats.classList.remove('hidden');
        });

        document.querySelectorAll('.close-modal').forEach(btn => {
            btn.addEventListener('click', () => {
                modalHelp.classList.add('hidden');
                modalStats.classList.add('hidden');
            });
        });

        // Cerrar modales al hacer clic en el fondo oscuro
        [modalHelp, modalStats].forEach(modal => {
            modal.addEventListener('click', (e) => {
                if (e.target === modal) {
                    modal.classList.add('hidden');
                }
            });
        });

        // Reiniciar historial para pruebas
        document.getElementById('btn-reset-progress').addEventListener('click', () => {
            localStorage.removeItem(STORAGE_KEY);
            appState = {
                currentCaseId: 1,
                stats: { played: 0, won: 0, currentStreak: 0, maxStreak: 0 },
                casesProgress: {}
            };
            activeTabArea = 'anamnesis';
            questionFilterText = '';
            loadState();
            saveState();
            renderAll();
            modalStats.classList.add('hidden');
            showToast("Historial reiniciado correctamente.");
        });

        // Inicialización al cargar la página
        window.addEventListener('DOMContentLoaded', () => {
            loadState();
            renderAll();
        });
    </script>
</body>
